  set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
  set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
  if(LIVE_PEER_MBEDTLS4)
    # The official release includes the generated sources and PSA library.
    set(GEN_FILES OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(mbedtls
      URL https://github.com/Mbed-TLS/mbedtls/releases/download/mbedtls-4.1.2/mbedtls-4.1.2.tar.bz2
      URL_HASH SHA256=392215e5e4846df04e2f4c4516079303bd73edde837d8f3e68fb4d4d7ddf0126)
  else()
    FetchContent_Declare(mbedtls
      URL https://github.com/Mbed-TLS/mbedtls/releases/download/mbedtls-3.6.8/mbedtls-3.6.8.tar.bz2
      URL_HASH SHA256=bc54b818e87f49c0b3281dd85d24adaede0e63b4e822e3fbd49c005a38b9595b)
  endif()
  FetchContent_MakeAvailable(mbedtls)
  set(PEER_MBEDTLS_SOURCE_DIR "${mbedtls_SOURCE_DIR}")
  include("${CMAKE_CURRENT_SOURCE_DIR}/patches/mbedtls-empty-name.cmake")
  if(LIVE_PEER_MBEDTLS4)
    # Preserve our consumer target name while upstream separates TLS and PSA.
    add_library(MbedTLS::mbedcrypto ALIAS tfpsacrypto)
    set(peer_crypto_target tfpsacrypto)
    if(LIVE_PEER_MBEDTLS_SRTP)
      message(FATAL_ERROR "The libSRTP Mbed TLS AES adapter requires Mbed TLS 3; use the default libSRTP backend with 4")
    endif()
  else()
    set(peer_crypto_target mbedcrypto)
  endif()
  foreach(t mbedtls mbedx509 ${peer_crypto_target})
    target_compile_definitions(${t} PUBLIC MBEDTLS_SSL_DTLS_SRTP)
  endforeach()
endif()

option(LIVE_PEER_FUZZ "Build libFuzzer parser harness (Clang)" OFF)
if(LIVE_PEER_FUZZ)
  if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "LIVE_PEER_FUZZ requires Clang")
  endif()
  foreach(t live_peer mbedtls mbedx509 mbedcrypto srtp2 usrsctp)
    if(TARGET ${t})
      get_target_property(imported ${t} IMPORTED)
      if(NOT imported)
        target_compile_options(${t} PRIVATE -fsanitize=fuzzer-no-link)
      endif()
    endif()
  endforeach()
  add_executable(peer_fuzz tests/fuzz.c)
  add_executable(peer_fuzz_dtls tests/fuzz_dtls.c)
  add_executable(peer_fuzz_sctp tests/fuzz_sctp.c)
  foreach(t peer_fuzz peer_fuzz_dtls peer_fuzz_sctp)
    target_include_directories(${t} PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/include" "${usrsctp_SOURCE_DIR}/usrsctplib")
    target_compile_options(${t} PRIVATE -fsanitize=fuzzer)
    target_link_options(${t} PRIVATE -fsanitize=fuzzer)
    target_link_libraries(${t} PRIVATE live_peer MbedTLS::mbedtls MbedTLS::mbedx509 MbedTLS::mbedcrypto srtp2)
  endforeach()
endif()
