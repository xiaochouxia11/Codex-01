cmake_minimum_required(VERSION 3.16)
project(live_esp32_host LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

# Use precisely the same parser source as the firmware build.
set(IDF_PATH "$ENV{IDF_PATH}" CACHE PATH "ESP-IDF checkout")
set(CJSON_DIR "${IDF_PATH}/components/json/cJSON" CACHE PATH "cJSON source directory")
if(NOT EXISTS "${CJSON_DIR}/cJSON.c")
    message(FATAL_ERROR "Source ESP-IDF export.sh first (validated with v5.5.6)")
endif()
option(LIVE_SANITIZE "Enable host address and undefined-behavior sanitizers" ON)
if(LIVE_SANITIZE AND CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_library(cjson STATIC "${CJSON_DIR}/cJSON.c")
target_include_directories(cjson PUBLIC "${CJSON_DIR}")
add_library(live STATIC components/live/live.c)
target_include_directories(live PUBLIC components/live/include)
target_link_libraries(live PUBLIC cjson)
target_compile_options(live PRIVATE -Wall -Wextra -Werror -Wpedantic)
add_executable(live_tests tests/test_live.c tests/main.c tests/header_cpp.cpp)
target_link_libraries(live_tests PRIVATE live)
target_compile_options(live_tests PRIVATE -Wall -Wextra -Werror -Wpedantic)
enable_testing()
